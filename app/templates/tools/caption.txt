<script>
document.addEventListener("DOMContentLoaded", function() {
  const lazyImages = document.querySelectorAll("img.lazy");

  if ("IntersectionObserver" in window) {
    let observer = new IntersectionObserver((entries, observer) => {
      entries.forEach(entry => {
        if (entry.isIntersecting) {
          let img = entry.target;
          img.src = img.dataset.src;
          img.classList.remove("lazy");
          if (img.previousElementSibling) img.previousElementSibling.remove();
          observer.unobserve(img);
        }
      });
    });
    lazyImages.forEach(img => observer.observe(img));
  } else {
    lazyImages.forEach(img => img.src = img.dataset.src);
  }
});
</script>

document.getElementById("xPostForm").addEventListener("submit", async (e) => {
  e.preventDefault();

<script>
document.getElementById("generateBtn").addEventListener("click", async () => {
  const text = document.getElementById("postInput").value.trim();
  const tone = document.getElementById("tone").value;
  const format = document.getElementById("format").value;
  const avoidClickbait = document.getElementById("avoidClickbait").checked;
  const generateImage = document.getElementById("generateImage").checked;

  // Detect if this is a custom prompt
  const isCustomPrompt = /Viral Story Creator|Write a powerful|Create a story/i.test(text);

  const payload = {
    text,
    tone,
    mode: format,
    avoid_clickbait: avoidClickbait,
    generate_images: generateImage,
    custom_prompt: isCustomPrompt ? text : null
  };

  const response = await fetch("/api/generate_x_post", {
    method: "POST",
    headers: {"Content-Type": "application/json"},
    body: JSON.stringify(payload)
  });

  const result = await response.json();

  // Clear preview
  document.getElementById("captions").innerHTML = "";
  document.getElementById("thread").innerHTML = "";
  document.getElementById("replies").innerHTML = "";
  document.getElementById("metrics").innerHTML = "";

  // Captions
  result.captions.forEach(c => {
    const div = document.createElement("div");
    div.innerHTML = `<strong>[${c.style}]</strong>: ${c.text} <br/> 
                     <em>Confidence Score:</em> ${c.confidence_score} | 
                     <em>Likes:</em> ${c.predicted_engagement.likes}, 
                     <em>Retweets:</em> ${c.predicted_engagement.retweets}, 
                     <em>Replies:</em> ${c.predicted_engagement.replies} <br/>
                     <em>Best Post Time:</em> ${c.best_post_time} <br/>
                     ${c.image_url ? `<img src="${c.image_url}" width="200"/>` : ""}`;
    document.getElementById("captions").appendChild(div);
  });

  // Thread
  if (result.thread && result.thread.length > 0) {
    const threadDiv = document.getElementById("thread");
    threadDiv.innerHTML = `<h4>Thread:</h4>`;
    result.thread.forEach((t, i) => {
      const tDiv = document.createElement("div");
      tDiv.innerHTML = `<strong>Tweet ${i + 1}:</strong> ${t.text} <br/>
                        ${t.image_url ? `<img src="${t.image_url}" width="180"/>` : ""}`;
      threadDiv.appendChild(tDiv);
    });
  }

  // Replies
  if (result.replies && result.replies.length > 0) {
    const repliesDiv = document.getElementById("replies");
    repliesDiv.innerHTML = `<h4>Suggested Replies:</h4><ul>` + result.replies.map(r => `<li>${r}</li>`).join("") + `</ul>`;
  }

  // Overall engagement
  document.getElementById("metrics").innerHTML = `<strong>Overall Engagement Score:</strong> ${result.engagement_score}`;
});
</script>
