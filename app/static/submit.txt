// static/submit.js
import { uploadImagesBeforeSave } from "./media.js";
import { updateHiddenInput } from "./helper.js";
import { setSafeRedirect } from "./utils.js";

/**
 * Initialize form submission for drafts only
 * @param {HTMLElement} editor
 * @param {HTMLFormElement} form
 */
export function initSubmit(editor, form) {
  if (!editor || !form) return;

  const contentInput = document.getElementById("content-hidden");
  const draftBtn = document.getElementById("draftBtn");
  const contentInput = document.getElementById("content-hidden");
  const statusInput = document.getElementById("post-status");
  const scheduledInput = document.getElementById("scheduled_at");

  const draftBtn = document.getElementById("draftBtn");
  const confirmScheduleBtn = document.getElementById("confirmScheduleBtn");

  if (!contentInput || !statusInput) return;
  
  const minWords = window.rules?.min_words || 150;
  const postIdInput = form.querySelector('[name="post_id"]') || (() => {
    const hidden = document.createElement("input");
    hidden.type = "hidden";
    hidden.name = "post_id";
    form.appendChild(hidden);
    return hidden;
  })();

  function resetButtons() {
    draftBtn && (draftBtn.textContent = "Save Draft", draftBtn.disabled = false);
    confirmScheduleBtn && (confirmScheduleBtn.textContent = "Confirm Schedule", confirmScheduleBtn.disabled = false);
  }

  // -----------------------------
  // Draft button click
  // -----------------------------
  draftBtn?.addEventListener("click", async (e) => {
    e.preventDefault();
    const content = editor.innerHTML.trim();
    if (!content) return;

    draftBtn.textContent = "Saving…";
    draftBtn.disabled = true;

    try {
      await uploadImagesBeforeSave(editor);

      const data = new FormData();
      data.set("content", content);
      data.set("title", form.title?.value || "Untitled Draft");
      data.set("status", "draft");
      data.set("post_id", postIdInput.value || "");

      const res = await fetch("/post/draft", { method: "POST", body: data, credentials: "same-origin" });
      if (!res.ok) throw new Error(`HTTP error! status: ${res.status}`);

      const json = await res.json();
      if (json.post_id) postIdInput.value = json.post_id;

      draftBtn.textContent = "Draft Saved ✅";

      // Optional redirect after save
      setTimeout(() => {
        window.location.href = "/dashboard";
      }, 1000);

    } catch (err) {
      console.error("Draft save failed:", err);
      draftBtn.textContent = "Draft ❌";
    } finally {
      setTimeout(resetButtons, 1500);
    }
  });
  
  confirmScheduleBtn?.addEventListener("click", () => {
    if (!scheduledInput?.value) return showError("Please select a schedule date");

    const scheduledDate = new Date(scheduledInput.value);
    const now = new Date();
    if (scheduledDate <= now) return showError("The selected date must be in the future");

    statusInput.value = "scheduled";
    form.dataset.action = "schedule";
    form.requestSubmit();
  });

  // -----------------------------
  // Form submit handler (fallback)
  // -----------------------------
  form.addEventListener("submit", async (e) => {
    e.preventDefault();
    if (form.dataset.submitting === "true") return;
    form.dataset.submitting = "true";

    const action = form.dataset.action;

    try {
      setLoading(true);

      if (action === "draft" && draftBtn) {
        draftBtn.textContent = "Saving…";
        draftBtn.disabled = true;
      }
      
      if (action === "schedule" && confirmScheduleBtn) {
        confirmScheduleBtn.textContent = "Confirming…";
        confirmScheduleBtn.disabled = true;
      }

      if (!navigator.onLine) {
        throw new Error("Network unavailable");
      }

      await uploadImagesBeforeSave(editor);

      const content = editor.innerHTML.trim();
      if (!content) throw new Error("Post content cannot be empty");
      
      if (form.dataset.action === "scheduled" && !scheduledInput?.value) {
        throw new Error("Schedule date missing");
      }

      contentInput.value = content;
      updateHiddenInput(document.getElementById("tags-hidden"), window.selectedTags || []);

      form.submit();
    } catch (err) {
      console.error(err);
      if (draftBtn) { draftBtn.textContent = "Draft ❌"; draftBtn.disabled = false; }
      alert(err.message || "Submission failed");
    } finally {
      delete form.dataset.action;
      form.dataset.submitting = "false";
      resetButtons();
    }
  });
